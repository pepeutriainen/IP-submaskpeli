        function setupOctetInputs() {
            const handleInput = (e) => {
                let target = e.target;
                // Vain numeroita
                target.value = target.value.replace(/[^0-9]/g, '');
                
                // Rajoita arvo 255
                if (target.value !== '' && parseInt(target.value) > 255) {
                    target.value = '255';
                }

                // Automaattinen siirtyminen seuraavaan kun 3 numeroa täynnä
                if (target.value.length === 3) {
                    const next = target.nextElementSibling?.nextElementSibling;
                    if (next && next.tagName === 'INPUT') next.focus();
                }
            };

            const handleKeyDown = (e) => {
                const target = e.target;
                // Siirry seuraavaan pisteellä tai välilyönnillä
                if ((e.key === '.' || e.key === ' ' || e.key === 'ArrowRight') && target.value !== '') {
                    e.preventDefault();
                    const next = target.nextElementSibling?.nextElementSibling;
                    if (next && next.tagName === 'INPUT') next.focus();
                } 
                // Peruuta edelliseen jos ruutu on tyhjä ja painetaan backspace
                else if ((e.key === 'Backspace' || e.key === 'ArrowLeft') && target.value === '') {
                    e.preventDefault();
                    const prev = target.previousElementSibling?.previousElementSibling;
                    if (prev && prev.tagName === 'INPUT') {
                        prev.focus();
                        // Vie kursori edellisen syötteen perään
                        setTimeout(() => prev.setSelectionRange(prev.value.length, prev.value.length), 0);
                    }
                }
            };

            document.querySelectorAll('.ip-octet, .mask-octet').forEach(input => {
                input.addEventListener('input', handleInput);
                input.addEventListener('keydown', handleKeyDown);
            });
        }

        // IP Matematiikka apufunktiot
        function ip2long(ip) {
            return ip.split('.').reduce((acc, octet) => (acc << 8) + parseInt(octet, 10), 0) >>> 0;
        }
        function long2ip(long) {
            return [ (long >>> 24), (long >> 16 & 255), (long >> 8 & 255), (long & 255) ].join('.');
        }
        function calculateSubnetDetails(networkIp, cidr) {
            const netLong = ip2long(networkIp);
            const mask = ~((1 << (32 - cidr)) - 1) >>> 0;
            const networkAddress = netLong & mask;
            const broadcastAddress = networkAddress | (~mask >>> 0);
            return {
                network: long2ip(networkAddress),
                broadcast: long2ip(broadcastAddress),
                firstHost: long2ip(networkAddress + 1),
                lastHost: long2ip(broadcastAddress - 1),
                mask: long2ip(mask)
            };
        }

        function initThreeJS() {
            const canvas = document.getElementById('game-canvas');
            scene = new THREE.Scene();
            
            // Isometrinen kamera
            const aspect = window.innerWidth / window.innerHeight;
            const d = 20;
            camera = new THREE.OrthographicCamera(-d * aspect, d * aspect, d, -d, 1, 1000);
            camera.position.set(20, 20, 20);
            camera.lookAt(scene.position);

            renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: true, alpha: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.setPixelRatio(window.devicePixelRatio);

            // Valot
            const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
            scene.add(ambientLight);
            const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
            dirLight.position.set(20, 40, 20);
            scene.add(dirLight);

            // Grid
            const gridHelper = new THREE.GridHelper(100, 50, 0x334155, 0x1e293b);
            gridHelper.position.y = -0.5;
            scene.add(gridHelper);

            raycaster = new THREE.Raycaster();
            mouse = new THREE.Vector2();

            // Event listenerit
            renderer.domElement.addEventListener('pointermove', onPointerMove);
            renderer.domElement.addEventListener('pointerdown', onPointerDown);
            window.addEventListener('pointerup', () => { isPanning = false; });
            window.addEventListener('resize', onWindowResize);
            renderer.domElement.addEventListener('contextmenu', (e) => e.preventDefault());
            
            renderer.domElement.addEventListener('wheel', (e) => {
                e.preventDefault();
                cameraZoom += e.deltaY * -0.001;
                cameraZoom = Math.max(0.3, Math.min(cameraZoom, 3));
                camera.zoom = cameraZoom;
                camera.updateProjectionMatrix();
            }, { passive: false });

            // Näppäimistö navigointi
            window.addEventListener('keydown', (e) => {
                if (e.key === 'Escape') {
                    if (cableActionState.active && cableActionState.lineTemp) {
                        scene.remove(cableActionState.lineTemp);
                        cableActionState = { active: false, startNode: null, lineTemp: null };
                        showToast("Toiminto peruttu", "info");
                    }
                    if (!document.getElementById('ip-modal').classList.contains('hidden')) closeIpModal();
                }
                const key = e.key.toLowerCase();
                if (keys.hasOwnProperty(key)) keys[key] = true;
                if (e.key === 'ArrowUp') keys.up = true;
                if (e.key === 'ArrowDown') keys.down = true;
                if (e.key === 'ArrowLeft') keys.left = true;
                if (e.key === 'ArrowRight') keys.right = true;
            });
            window.addEventListener('keyup', (e) => {
                const key = e.key.toLowerCase();
                if (keys.hasOwnProperty(key)) keys[key] = false;
                if (e.key === 'ArrowUp') keys.up = false;
                if (e.key === 'ArrowDown') keys.down = false;
                if (e.key === 'ArrowLeft') keys.left = false;
                if (e.key === 'ArrowRight') keys.right = false;
            });

            animate();
        }

        function onWindowResize() {
            const aspect = window.innerWidth / window.innerHeight;
            const d = 20;
            camera.left = -d * aspect;
            camera.right = d * aspect;
            camera.top = d;
            camera.bottom = -d;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        function onPointerMove(event) {
            if (isPanning) {
                const dx = event.clientX - panStart.x;
                const dy = event.clientY - panStart.y;
                const panSpeed = 0.05 / cameraZoom;
                cameraTarget.x -= (dx + dy) * panSpeed;
                cameraTarget.z -= (dy - dx) * panSpeed;
                panStart.x = event.clientX;
                panStart.y = event.clientY;
                return;
            }

            mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
            mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

            if (cableActionState.active && cableActionState.lineTemp) {
                raycaster.setFromCamera(mouse, camera);
                const intersects = raycaster.intersectObject(scene.children.find(c => c.type === 'GridHelper'));
                if (intersects.length > 0) {
                    const pos = intersects[0].point;
                    const positions = cableActionState.lineTemp.geometry.attributes.position.array;
                    positions[3] = pos.x; positions[4] = 0; positions[5] = pos.z;
                    cableActionState.lineTemp.geometry.attributes.position.needsUpdate = true;
                }
            }
        }

        function onPointerDown(event) {
            if(event.target.closest('.pointer-events-auto')) return;

            if (event.button === 2 || event.button === 1) {
                isPanning = true;
                panStart.x = event.clientX;
                panStart.y = event.clientY;
                return;
            }

            raycaster.setFromCamera(mouse, camera);
            
            // Tarkista osuttiinko nodeen
            const interactableObjects = nodes.map(n => n.mesh);
            const intersects = raycaster.intersectObjects(interactableObjects);

            if (intersects.length > 0) {
                const clickedMesh = intersects[0].object;
                const clickedNode = nodes.find(n => n.mesh === clickedMesh);

                if (currentTool === 'select') {
                    if ([nodeTypes.OFFICE, nodeTypes.PC, nodeTypes.SERVER].includes(clickedNode.userData.type)) {
                        if (clickedNode.userData.isConnected) {
                            openIpModal(clickedNode);
                        } else {
                            showToast("Laitteella ei ole verkkoyhteyttä reitittimeen!", "error");
                        }
                    } else {
                        showToast(`Tyyppi: ${clickedNode.userData.type.toUpperCase()}`, "info");
                    }
                } else if (currentTool === 'cable') {
                    handleCableTool(clickedNode);
                } else if (currentTool === 'delete') {
                    if (clickedNode.userData.isPredefined) {
                        showToast("Et voi poistaa tason kiinteitä kohteita!", "error");
                    } else {
                        deleteNode(clickedNode);
                    }
                }
            } else {
                // Klikattiin tyhjää - rakennetaan uusi laite jos työkalu sallii
                if (currentTool === 'switch' || currentTool === 'wifi') {
                    const gridIntersects = raycaster.intersectObject(scene.children.find(c => c.type === 'GridHelper'));
                    if (gridIntersects.length > 0) {
                        const point = gridIntersects[0].point;
                        // Snap to grid
                        const x = Math.round(point.x / 2) * 2;
                        const z = Math.round(point.z / 2) * 2;
                        
                        // Estä päällekkäisyys
                        if (!nodes.find(n => Math.abs(n.mesh.position.x - x) < 1 && Math.abs(n.mesh.position.z - z) < 1)) {
                            createNode(currentTool, x, z, false);
                        }
                    }
                }
            }
        }

        function createNode(type, x, z, isPredefined = false) {
            let geometry, material, color;

            switch(type) {
                case nodeTypes.ROUTER:
                    geometry = new THREE.CylinderGeometry(1, 1, 1.5, 16);
                    color = 0x3b82f6; // Blue
                    break;
                case nodeTypes.SWITCH:
                    geometry = new THREE.BoxGeometry(1.5, 0.5, 1.5);
                    color = 0xf97316; // Orange
                    break;
                case nodeTypes.WIFI:
                    geometry = new THREE.SphereGeometry(0.8, 16, 16);
                    color = 0xa855f7; // Purple
                    break;
                case nodeTypes.OFFICE:
                    geometry = new THREE.BoxGeometry(2, 2, 2);
                    color = 0x64748b; // Gray
                    break;
                case nodeTypes.PC:
                    geometry = new THREE.BoxGeometry(1, 1, 1);
                    color = 0x94a3b8; // Light Gray
                    break;
                case nodeTypes.SERVER:
                    geometry = new THREE.BoxGeometry(1.2, 3, 1.2);
                    color = 0x1e293b; // Dark Gray
                    break;
            }

            material = new THREE.MeshPhongMaterial({ color: color });
            const mesh = new THREE.Mesh(geometry, material);
            
            // Asetetaan pohja nollaan
            mesh.geometry.computeBoundingBox();
            const height = mesh.geometry.boundingBox.max.y - mesh.geometry.boundingBox.min.y;
            mesh.position.set(x, height/2, z);

            scene.add(mesh);
            
            const node = { 
                id: Math.random().toString(36).substr(2, 9), 
                mesh: mesh, 
                userData: { 
                    type: type, 
                    isPredefined: isPredefined,
                    isConnected: false,
                    correctIp: false,
                    ip: null
                }
            };

            // Jos WiFi, piirretään kantama-alue
            if (type === nodeTypes.WIFI) {
                const ringGeo = new THREE.RingGeometry(14.5, 15, 32);
                const ringMat = new THREE.MeshBasicMaterial({ color: 0xa855f7, side: THREE.DoubleSide, transparent: true, opacity: 0.3 });
                const ring = new THREE.Mesh(ringGeo, ringMat);
                ring.rotation.x = Math.PI / 2;
                ring.position.y = 0.1;
                mesh.add(ring);
                node.userData.wifiRange = 15;
            }

            nodes.push(node);
            checkConnections();
            return node;
        }

        function deleteNode(node) {
            // Poista ensin kaikki nodeen liittyvät kaapelit
            cables = cables.filter(cable => {
                if (cable.nodeA === node || cable.nodeB === node) {
                    scene.remove(cable.line);
                    return false;
                }
                return true;
            });
            scene.remove(node.mesh);
            nodes = nodes.filter(n => n !== node);
            checkConnections();
        }

        function handleCableTool(node) {
            if (!cableActionState.active) {
                cableActionState.active = true;
                cableActionState.startNode = node;
                
                const material = new THREE.LineBasicMaterial({ color: 0xe2e8f0, linewidth: 2, dashed: true, dashSize: 0.5, gapSize: 0.5 });
                const geometry = new THREE.BufferGeometry().setFromPoints([node.mesh.position.clone(), node.mesh.position.clone()]);
                const line = new THREE.Line(geometry, material);
                line.computeLineDistances();
                scene.add(line);
                cableActionState.lineTemp = line;
            } else {
                if (cableActionState.startNode !== node) {
                    const dist = cableActionState.startNode.mesh.position.distanceTo(node.mesh.position);
                    // Pituusrajoitus (100 metriä = n. 15 yksikköä pelissä)
                    if (dist > 15) {
                        showToast("Kaapeli on liian pitkä! (Yli 100m). Signaali heikkenee. Käytä kytkintä välissä.", "error");
                        scene.remove(cableActionState.lineTemp);
                    } else {
                        // Varmista ettei kaapelia ole jo
                        const exists = cables.find(c => 
                            (c.nodeA === cableActionState.startNode && c.nodeB === node) || 
                            (c.nodeB === cableActionState.startNode && c.nodeA === node)
                        );
                        if (!exists) {
                            const material = new THREE.LineBasicMaterial({ color: 0x94a3b8, linewidth: 3 });
                            const geometry = new THREE.BufferGeometry().setFromPoints([cableActionState.startNode.mesh.position.clone(), node.mesh.position.clone()]);
                            const line = new THREE.Line(geometry, material);
                            scene.add(line);
                            cables.push({ nodeA: cableActionState.startNode, nodeB: node, line: line });
                            
                            createSparks(node.mesh.position.clone());
                        }
                    }
                } else {
                    scene.remove(cableActionState.lineTemp);
                }
                if(cableActionState.lineTemp) scene.remove(cableActionState.lineTemp);
                cableActionState = { active: false, startNode: null, lineTemp: null };
                checkConnections();
            }
        }

        function checkConnections() {
            // Nollaa tila
            nodes.forEach(n => {
                if (n.userData.type !== nodeTypes.ROUTER) {
                    n.userData.isConnected = false;
                    // Palauta väri jos ei oikeaa IP:tä
                    if (!n.userData.correctIp && n.userData.type !== nodeTypes.SWITCH && n.userData.type !== nodeTypes.WIFI) {
                        n.mesh.material.color.setHex(n.userData.type === nodeTypes.PC ? 0x94a3b8 : (n.userData.type === nodeTypes.SERVER ? 0x1e293b : 0x64748b));
                    }
                }
            });

            const router = nodes.find(n => n.userData.type === nodeTypes.ROUTER);
            if (!router) return;

            router.userData.isConnected = true;

            // 1. BFS Kaapeleita pitkin
            let queue = [router];
            let visited = new Set();
            visited.add(router);

            // Rakenna vieruslista
            let adj = new Map();
            nodes.forEach(n => adj.set(n, []));
            cables.forEach(c => {
                adj.get(c.nodeA).push(c.nodeB);
                adj.get(c.nodeB).push(c.nodeA);
            });

            while (queue.length > 0) {
                const curr = queue.shift();
                curr.userData.isConnected = true;

                for (let neighbor of adj.get(curr)) {
                    if (!visited.has(neighbor)) {
                        visited.add(neighbor);
                        queue.push(neighbor);
                    }
                }
            }

            // 2. WiFi yhteydet
            const activeWifis = nodes.filter(n => n.userData.type === nodeTypes.WIFI && n.userData.isConnected);
            nodes.forEach(node => {
                if ([nodeTypes.OFFICE, nodeTypes.PC, nodeTypes.SERVER].includes(node.userData.type) && !node.userData.isConnected) {
                    for (let wifi of activeWifis) {
                        if (node.mesh.position.distanceTo(wifi.mesh.position) <= wifi.userData.wifiRange) {
                            node.userData.isConnected = true;
                            // Visuaalinen indikaattori wifi yhteydestä voisi olla kiva, esim viiva
                            break;
                        }
                    }
                }
            });

            updateGoalUI();
        }

        function createZone(x, z, w, d, colorHex) {
            const geometry = new THREE.PlaneGeometry(w, d);
            const material = new THREE.MeshBasicMaterial({ 
                color: colorHex, 
                transparent: true, 
                opacity: 0.15, 
                side: THREE.DoubleSide,
                depthWrite: false
            });
            const mesh = new THREE.Mesh(geometry, material);
            mesh.rotation.x = -Math.PI / 2;

            mesh.add(line);

            scene.add(mesh);
            zoneMeshes.push(mesh);
        }

        function submitIp() {
            // Kerää osat lokeroista
            const ipParts = Array.from(document.querySelectorAll('.ip-octet')).map(el => el.value.trim());
            const maskParts = Array.from(document.querySelectorAll('.mask-octet')).map(el => el.value.trim());

            if (ipParts.some(p => p === '') || maskParts.some(p => p === '')) {
                showToast("Täytä IP-osoitteen kaikki neljä osaa!", "error");
                return;
            }

            const ip = ipParts.join('.');
            const mask = maskParts.join('.');

            // Validoi IPv4 muoto (Tämä on osin turha koska lokerot sallivat vain numeroita 0-255, mutta varmuuden vuoksi)
            const ipRegex = /^(\d{1,3}\.){3}\d{1,3}$/;
            if (!ipRegex.test(ip) || !ipRegex.test(mask)) {
                showToast("Virheellinen IP- tai peitemuoto!", "error");
                return;
            }

            const numIpParts = ip.split('.').map(Number);
            const numMaskParts = mask.split('.').map(Number);
            if (numIpParts.some(p => p > 255) || numMaskParts.some(p => p > 255)) {
                showToast("Oktetti ei voi olla yli 255!", "error");
                return;
            }

            const details = currentLevelConfig.subnetDetails;

            // Opetuksellinen osuus: Tarkista ensin Subnet Mask
            if (mask !== details.mask) {
                showToast(`Väärä aliverkon peite! /${currentLevelConfig.cidr} verkossa peite on ${details.mask}`, "error");
                return;
            }

            // Tarkista verkon logiikka IP:lle
            if (ip === details.network) {
                showToast(`Virhe: ${ip} on verkon osoite! Laitteelle pitää antaa isäntäosoite.`, "error");
            } else if (ip === details.broadcast) {
                showToast(`Virhe: ${ip} on Broadcast-osoite! Sitä ei voi antaa laitteelle.`, "error");
            } else {
                const ipL = ip2long(ip);
                const firstL = ip2long(details.firstHost);
                const lastL = ip2long(details.lastHost);

                if (ipL >= firstL && ipL <= lastL) {
                    // Oikein!
                    selectedNodeForIp.userData.ip = ip;
                    selectedNodeForIp.userData.mask = mask;
                    selectedNodeForIp.userData.correctIp = true;
                    selectedNodeForIp.mesh.material.color.setHex(0x10b981); // Vihreä
                    
                    showToast("IP hyväksytty! Laite on verkossa.", "success");
                    
                    // Juhla-animaatio
                    selectedNodeForIp.userData.animating = true;
                    selectedNodeForIp.userData.animStart = Date.now();
                    createDataConfetti(selectedNodeForIp.mesh.position.clone());

                    closeIpModal();
                    updateGoalUI();
                } else {
                    showToast(`IP ei kuulu verkkoon ${details.network}/${currentLevelConfig.cidr}`, "error");
                }
            }
        }

        function updateGoalUI() {
            const required = currentLevelConfig ? currentLevelConfig.requiredNodes.length : 0;

            // Tarkista kaikki tehtäväksi annetut (predefined) laitteet, jotka EIVÄT ole reitittimiä
            nodes.forEach(n => {
                if (n.userData.isPredefined && n.userData.type !== nodeTypes.ROUTER) {
                    if (n.userData.isConnected && n.userData.correctIp) {
                        current++;
                    }
                }
            });

            document.getElementById('goal-current').innerText = current;
            
            const goalCurrentEl = document.getElementById('goal-current');

                goalCurrentEl.classList.remove('text-white');
                goalCurrentEl.classList.add('text-green-400');

                goalCurrentEl.classList.remove('text-green-400');
            }
        }

        function createSparks(position) {
            for (let i = 0; i < 15; i++) {
                const geo = new THREE.BoxGeometry(0.2, 0.2, 0.2);
                const mat = new THREE.MeshBasicMaterial({ color: 0x60a5fa });
                const mesh = new THREE.Mesh(geo, mat);
                mesh.position.copy(position);
                mesh.position.y += 1;
                scene.add(mesh);
                particles.push({
                    mesh: mesh,
                    velocity: new THREE.Vector3((Math.random()-0.5)*0.5, Math.random()*0.5, (Math.random()-0.5)*0.5),
                    life: 1.0
                });
            }
        }

        function createDataConfetti(position) {
            for (let i = 0; i < 30; i++) {
                const geo = new THREE.BoxGeometry(0.3, 0.3, 0.3);
                const mat = new THREE.MeshBasicMaterial({ color: 0x10b981 });
                const mesh = new THREE.Mesh(geo, mat);
                mesh.position.copy(position);
                mesh.position.y += 2;
                scene.add(mesh);
                particles.push({
                    mesh: mesh,
                    velocity: new THREE.Vector3((Math.random()-0.5)*0.8, Math.random()*0.8 + 0.2, (Math.random()-0.5)*0.8),
                    life: 1.5,
                    isConfetti: true
                });
            }
        }

        function animate() {
            requestAnimationFrame(animate);

            // Kameran ohjaus näppäimistöllä
            if (!document.getElementById('game-ui').classList.contains('hidden')) {
                const speed = 0.5;
                if (keys.w || keys.up) { cameraTarget.x -= speed; cameraTarget.z -= speed; }
                if (keys.s || keys.down) { cameraTarget.x += speed; cameraTarget.z += speed; }
                if (keys.a || keys.left) { cameraTarget.x -= speed; cameraTarget.z += speed; }
                if (keys.d || keys.right) { cameraTarget.x += speed; cameraTarget.z -= speed; }
            }

            // Lerp kamera kohteeseen
            if (document.getElementById('main-menu').classList.contains('hidden')) {
                camera.position.x = THREE.MathUtils.lerp(camera.position.x, cameraTarget.x + 20, 0.1);
                camera.position.z = THREE.MathUtils.lerp(camera.position.z, cameraTarget.z + 20, 0.1);
                camera.lookAt(cameraTarget.x, 0, cameraTarget.z);
            } else {
                // Valikossa pyöritään
                const time = Date.now() * 0.0001;
                camera.position.x = Math.cos(time) * 30;
                camera.position.z = Math.sin(time) * 30;
                camera.lookAt(0,0,0);
            }

            // Partikkelit
            for (let i = particles.length - 1; i >= 0; i--) {
                const p = particles[i];
                p.mesh.position.add(p.velocity);
                p.velocity.y -= 0.02; // Painovoima
                p.life -= 0.02;
                
                if (p.isConfetti) p.mesh.rotation.x += 0.1; p.mesh.rotation.y += 0.1;

                if (p.life <= 0) {
                    scene.remove(p.mesh);
                    particles.splice(i, 1);
                } else {
                    p.mesh.scale.setScalar(p.life);
                }
            }

            // Pomppivat laitteet (Onnistuminen)
            nodes.forEach(n => {
                if (n.userData.animating) {
                    const elapsed = Date.now() - n.userData.animStart;
                    if (elapsed < 500) {
                        // Hyppy (Siniaalto)
                        n.mesh.position.y = (n.mesh.geometry.boundingBox.max.y - n.mesh.geometry.boundingBox.min.y)/2 + Math.sin(elapsed / 500 * Math.PI) * 2;
                    } else {
                        n.mesh.position.y = (n.mesh.geometry.boundingBox.max.y - n.mesh.geometry.boundingBox.min.y)/2;
                        n.userData.animating = false;
                    }
                }
            });

            renderer.render(scene, camera);
        }

        window.onload = () => {
            generateLevels();
            renderLevelMenu();
            setupTools();
            setupOctetInputs(); // Uusi IP lokeroiden alustus
            initThreeJS();

            document.getElementById('btn-back-menu').addEventListener('click', goToMenu);
            document.getElementById('btn-reset-progress').addEventListener('click', () => {
                if(confirm("Haluatko varmasti nollata edistymisesi?")) {
                    localStorage.removeItem('subnetArchitect_unlocked');
                    unlockedLevels = 1;
                    renderLevelMenu();
                }
            });
        };
